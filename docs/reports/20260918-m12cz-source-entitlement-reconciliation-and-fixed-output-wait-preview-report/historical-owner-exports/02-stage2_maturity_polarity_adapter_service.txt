   42  def _canonical_sha256(value: object) -> str:
   43      payload = json.dumps(value, ensure_ascii=True, sort_keys=True, separators=(",", ":"))
   44      return hashlib.sha256(payload.encode()).hexdigest()

   47  def maturity_atomic_claim_ref(*, ticker: str, claim: EvidenceClaim) -> str:
   48      """Identify a structured proposition independently from its assigned polarity."""
   49      identity = {
   50          "contract": ATOMIC_IDENTITY_CONTRACT,
   51          "ticker": ticker,
   52          "text": claim.text,
   53          "evidence_refs": sorted(claim.evidence_refs),
   54          "logical_condition": (
   55              claim.logical_condition.model_dump(mode="json")

   58          ),
   59      }
   60      return CLAIM_REF_PREFIX + _canonical_sha256(identity)

   83  def stage2_maturity_atomic_claim_catalog(
   84      *,
   85      ticker: str,
   86      buy_drivers: Sequence[EvidenceClaim],
   87      sell_drivers: Sequence[EvidenceClaim],
   88  ) -> tuple[MaturityAtomicClaim, ...]:
   89      """Project frozen structured drivers through the existing polarity vocabulary."""
   90      projected = (
   91          *(
   92              _owned_claim(ticker=ticker, claim=claim, polarity=EvidencePolarity.BULLISH)
   93              for claim in buy_drivers
   94          ),
   95          *(
   96              _owned_claim(ticker=ticker, claim=claim, polarity=EvidencePolarity.BEARISH)
   97              for claim in sell_drivers
   98          ),
   99      )
  100      by_ref: dict[str, MaturityAtomicClaim] = {}
  101      for row in projected:
  102          existing = by_ref.get(row.claim_ref)
  103          if existing is not None and existing != row:
  104              raise ValueError(f"maturity_atomic_claim_identity_conflict:{row.claim_ref}")
  105          by_ref[row.claim_ref] = row
  106      return tuple(by_ref[key] for key in sorted(by_ref))
